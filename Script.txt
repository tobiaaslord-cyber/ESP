local function updateInstinctFilters()
	local filter = {}

	if LocalPlayer.Character then
		table.insert(filter, LocalPlayer.Character)
	end

	for _, part in ipairs(InstinctParts) do
		if part and part.Parent then
			table.insert(filter, part)
		end
	end

	InstinctOverlapParams.FilterDescendantsInstances = filter
end

local function updateInstinctSpaceFilters(threatModel)
	local filter = {}

	if LocalPlayer.Character then
		table.insert(filter, LocalPlayer.Character)
	end

	if threatModel then
		table.insert(filter, threatModel)
	end

	for _, part in ipairs(InstinctParts) do
		if part and part.Parent then
			table.insert(filter, part)
		end
	end

	InstinctSpaceParams.FilterDescendantsInstances = filter
end

local function destroyInstinctRing()
	for _, part in ipairs(InstinctParts) do
		if part and part.Parent then
			part:Destroy()
		end
	end

	table.clear(InstinctParts)
end

local function createInstinctRing()
	if #InstinctParts > 0 then
		return
	end

	for i = 1, INSTINCT_SEGMENTS do
		local part = Instance.new("Part")

		part.Name = "InstinctRingSegment"
		part.Anchored = true
		part.CanCollide = false
		part.CanTouch = false
		part.CanQuery = false
		part.CastShadow = false
		part.Material = Enum.Material.Neon
		part.Color = Color3.fromRGB(95, 125, 255)
		part.Transparency = 0.3
		part.Size = Vector3.new(0.11, 0.08, 1)
		part.TopSurface = Enum.SurfaceType.Smooth
		part.BottomSurface = Enum.SurfaceType.Smooth
		part.Parent = workspace

		table.insert(
			InstinctParts,
			part
		)
	end

	updateInstinctFilters()
end

local function updateInstinctRing()
	if not SETTINGS.Instinct then
		if #InstinctParts > 0 then
			destroyInstinctRing()
		end

		return
	end

	if #InstinctParts == 0 then
		createInstinctRing()
	end

	local root = getRoot(
		LocalPlayer.Character
	)

	if not root then
		return
	end

	local center =
		root.Position +
		Vector3.new(0, 0.07, 0)

	local radius =
		SETTINGS.InstinctRadius

	for i, part in ipairs(
		InstinctParts
	) do
		local angle1 =
			(
				(i - 1) /
				INSTINCT_SEGMENTS
			) *
			math.pi *
			2

		local angle2 =
			(
				i /
				INSTINCT_SEGMENTS
			) *
			math.pi *
			2

		local p1 =
			center +
			Vector3.new(
				math.cos(angle1) *
					radius,
				0,
				math.sin(angle1) *
					radius
			)

		local p2 =
			center +
			Vector3.new(
				math.cos(angle2) *
					radius,
				0,
				math.sin(angle2) *
					radius
			)

		local difference =
			p2 - p1

		part.Size =
			Vector3.new(
				0.11,
				0.08,
				difference.Magnitude
			)

		part.CFrame =
			CFrame.lookAt(
				p1 +
					difference *
					0.5,
				p2
			)
	end
end

local function getThreatRoot(model)
	if not model
		or not model:IsA("Model")
		or model == LocalPlayer.Character then

		return nil
	end

	local player =
		Players:GetPlayerFromCharacter(
			model
		)

	if player
		and isTeammate(player) then
		return nil
	end

	local humanoid =
		getHumanoid(model)

	local root =
		getRoot(model)

	if humanoid
		and root
		and humanoid.Health > 0 then

		return root
	end

	return nil
end

local function findClosestThreat()
	local localRoot =
		getRoot(
			LocalPlayer.Character
		)

	if not localRoot then
		return nil
	end

	local radius =
		SETTINGS.InstinctRadius +
		3

	local closestRoot
	local closestDistance =
		radius

	local checkedCharacters = {}

	for _, player in ipairs(
		Players:GetPlayers()
	) do
		if player ~= LocalPlayer
			and not isTeammate(player) then

			local character =
				player.Character

			local root =
				getThreatRoot(character)

			if root then
				local offset =
					root.Position -
					localRoot.Position

				local horizontal =
					Vector3.new(
						offset.X,
						0,
						offset.Z
					)

				local distance =
					horizontal.Magnitude

				checkedCharacters[
					character
				] = true

				if distance <= closestDistance then
					closestDistance =
						distance

					closestRoot =
						root
				end
			end
		end
	end

	updateInstinctFilters()

	local parts =
		workspace:GetPartBoundsInRadius(
			localRoot.Position,
			radius,
			InstinctOverlapParams
		)

	local checkedModels = {}

	for _, part in ipairs(parts) do
		local model =
			part:FindFirstAncestorOfClass(
				"Model"
			)

		if model
			and not checkedModels[model]
			and not checkedCharacters[model]
			and model ~= LocalPlayer.Character then

			checkedModels[model] = true

			local root =
				getThreatRoot(model)

			if root then
				local offset =
					root.Position -
					localRoot.Position

				local horizontal =
					Vector3.new(
						offset.X,
						0,
						offset.Z
					)

				local distance =
					horizontal.Magnitude

				if distance <= closestDistance then
					closestDistance =
						distance

					closestRoot =
						root
				end
			end
		end
	end

	return closestRoot
end

local function hasClearPath(
	origin,
	direction,
	distance,
	character,
	threatModel
)
	local rayParams =
		RaycastParams.new()

	rayParams.FilterType =
		Enum.RaycastFilterType.Exclude

	rayParams.FilterDescendantsInstances = {
		character,
		threatModel
	}

	rayParams.IgnoreWater = true

	for _, height in ipairs({
		0.5,
		1.8,
		3.2
	}) do
		if workspace:Raycast(
			origin +
				Vector3.new(
					0,
					height,
					0
				),
			direction *
				distance,
			rayParams
		) then
			return false
		end
	end

	return true
end

local function isTargetSpaceClear(
	target,
	character,
	threatModel
)
	updateInstinctSpaceFilters(
		threatModel
	)

	local parts =
		workspace:GetPartBoundsInBox(
			CFrame.new(
				target +
					Vector3.new(
						0,
						2.2,
						0
					)
			),
			Vector3.new(
				3.6,
				4.2,
				3.6
			),
			InstinctSpaceParams
		)

	return #parts == 0
end

local function getGroundedTarget(
	position,
	character,
	threatModel
)
	local rayParams =
		RaycastParams.new()

	rayParams.FilterType =
		Enum.RaycastFilterType.Exclude

	rayParams.FilterDescendantsInstances = {
		character,
		threatModel
	}

	rayParams.IgnoreWater = true

	local result =
		workspace:Raycast(
			position +
				Vector3.new(
					0,
					10,
					0
				),
			Vector3.new(
				0,
				-35,
				0
			),
			rayParams
		)

	if result then
		return Vector3.new(
			position.X,
			result.Position.Y +
				3,
			position.Z
		)
	end

	return position
end

local function makeEscapeCandidate(
	origin,
	direction,
	distance,
	character,
	threatRoot
)
	if direction.Magnitude < 0.001 then
		return nil
	end

	direction =
		Vector3.new(
			direction.X,
			0,
			direction.Z
		)

	if direction.Magnitude < 0.001 then
		return nil
	end

	direction = direction.Unit

	local threatModel =
		threatRoot and
		threatRoot.Parent

	if not hasClearPath(
		origin,
		direction,
		distance,
		character,
		threatModel
	) then
		return nil
	end

	local target =
		getGroundedTarget(
			origin +
				direction *
				distance,
			character,
			threatModel
		)

	if not isTargetSpaceClear(
		target,
		character,
		threatModel
	) then
		return nil
	end

	return target
end

local function getEscapePosition(threatRoot)
	local character =
		LocalPlayer.Character

	local localRoot =
		getRoot(character)

	if not character
		or not localRoot
		or not threatRoot then

		return nil
	end

	local origin =
		localRoot.Position

	local away =
		origin -
		threatRoot.Position

	local horizontalAway =
		Vector3.new(
			away.X,
			0,
			away.Z
		)

	if horizontalAway.Magnitude < 0.05 then
		horizontalAway =
			Vector3.new(
				-localRoot.CFrame.LookVector.X,
				0,
				-localRoot.CFrame.LookVector.Z
			)
	end

	if horizontalAway.Magnitude < 0.05 then
		horizontalAway =
			Vector3.new(
				0,
				0,
				1
			)
	end

	local forward =
		Vector3.new(
			localRoot.CFrame.LookVector.X,
			0,
			localRoot.CFrame.LookVector.Z
		)

	local right =
		Vector3.new(
			localRoot.CFrame.RightVector.X,
			0,
			localRoot.CFrame.RightVector.Z
		)

	if forward.Magnitude > 0 then
		forward = forward.Unit
	end

	if right.Magnitude > 0 then
		right = right.Unit
	end

	local threatDirection =
		Vector3.new(
			threatRoot.Position.X -
				origin.X,
			0,
			threatRoot.Position.Z -
				origin.Z
		)

	if threatDirection.Magnitude > 0 then
		threatDirection =
			threatDirection.Unit
	else
		threatDirection =
			forward
	end

	local candidates = {}

	local directTarget =
		makeEscapeCandidate(
			origin,
			horizontalAway,
			INSTINCT_ESCAPE_DISTANCE,
			character,
			threatRoot
		)

	if directTarget then
		table.insert(
			candidates,
			{
				position = directTarget,
				score =
					(
						directTarget -
						threatRoot.Position
					).Magnitude +
					100
			}
		)
	end

	local leftTarget =
		makeEscapeCandidate(
			origin,
			-right,
			INSTINCT_SIDE_DISTANCE,
			character,
			threatRoot
		)

	if leftTarget then
		table.insert(
			candidates,
			{
				position = leftTarget,
				score =
					(
						leftTarget -
						threatRoot.Position
					).Magnitude +
					40
			}
		)
	end

	local rightTarget =
		makeEscapeCandidate(
			origin,
			right,
			INSTINCT_SIDE_DISTANCE,
			character,
			threatRoot
		)

	if rightTarget then
		table.insert(
			candidates,
			{
				position = rightTarget,
				score =
					(
						rightTarget -
						threatRoot.Position
					).Magnitude +
					40
			}
		)
	end

	if #candidates > 0 then
		table.sort(
			candidates,
			function(a, b)
				return a.score > b.score
			end
		)

		return candidates[1].position
	end

	for _, direction in ipairs({
		threatDirection,
		forward
	}) do
		for _, distance in ipairs({
			INSTINCT_BEHIND_DISTANCE,
			INSTINCT_BEHIND_DISTANCE + 5,
			INSTINCT_BEHIND_DISTANCE + 10,
			INSTINCT_BEHIND_DISTANCE + 15
		}) do
			local target =
				makeEscapeCandidate(
					origin,
					direction,
					(
						origin -
						threatRoot.Position
					).Magnitude +
						distance,
					character,
					threatRoot
				)

			if target then
				return target
			end
		end
	end

	return nil
end

local function triggerInstinct()
	if not SETTINGS.Instinct then
		return
	end

	local character =
		LocalPlayer.Character

	local localRoot =
		getRoot(character)

	local humanoid =
		getHumanoid(character)

	if not character
		or not localRoot
		or not humanoid
		or humanoid.Health <= 0 then

		return
	end

	local now = os.clock()

	if now - LastInstinctTrigger <
		INSTINCT_COOLDOWN then

		return
	end

	local didTeleport = false

	for _ = 1, INSTINCT_CHAIN_ATTEMPTS do
		local threatRoot =
			findClosestThreat()

		if not threatRoot then
			break
		end

		local currentRoot =
			getRoot(character)

		if not currentRoot then
			break
		end

		local offset =
			threatRoot.Position -
			currentRoot.Position

		local horizontal =
			Vector3.new(
				offset.X,
				0,
				offset.Z
			)

		if horizontal.Magnitude >
			SETTINGS.InstinctRadius then

			break
		end

		local targetPosition =
			getEscapePosition(
				threatRoot
			)

		if not targetPosition then
			break
		end

		local rotation =
			currentRoot.CFrame -
			currentRoot.Position

		character:PivotTo(
			CFrame.new(
				targetPosition
			) *
			rotation
		)

		didTeleport = true

		task.wait()
	end

	if didTeleport then
		LastInstinctTrigger = now
	end
end

local function isVFXObject(object)
	return object:IsA("ParticleEmitter")
		or object:IsA("Beam")
		or object:IsA("Trail")
		or object:IsA("Fire")
		or object:IsA("Smoke")
		or object:IsA("Sparkles")
end

local function saveVFXOriginal(object)
	if not object
		or not object.Parent
		or not isVFXObject(object)
		or VFXInstances[object] then

		return
	end

	if object:IsA("ParticleEmitter")
		or object:IsA("Beam")
		or object:IsA("Trail") then

		VFXInstances[object] = {
			Type = object.ClassName,
			Color = object.Color
		}
	elseif object:IsA("Fire") then
		VFXInstances[object] = {
			Type = "Fire",
			Color = object.Color,
			SecondaryColor =
				object.SecondaryColor
		}
	elseif object:IsA("Smoke") then
		VFXInstances[object] = {
			Type = "Smoke",
			Color = object.Color
		}
	elseif object:IsA("Sparkles") then
		VFXInstances[object] = {
			Type = "Sparkles",
			Color = object.SparkleColor
		}
	end
end

local function setVFXColor(
	object,
	color
)
	if not object
		or not object.Parent
		or not isVFXObject(object) then

		return
	end

	saveVFXOriginal(object)

	if object:IsA("ParticleEmitter")
		or object:IsA("Beam")
		or object:IsA("Trail") then

		object.Color =
			ColorSequence.new(
				color
			)
	elseif object:IsA("Fire") then
		object.Color = color
		object.SecondaryColor =
			color:Lerp(
				Color3.new(
					1,
					1,
					1
				),
				0.35
			)
	elseif object:IsA("Smoke") then
		object.Color = color
	elseif object:IsA("Sparkles") then
		object.SparkleColor = color
	end
end

local function restoreVFXColor(object)
	local original =
		VFXInstances[object]

	if not original
		or not object
		or not object.Parent then

		return
	end

	if original.Type ==
		"ParticleEmitter"
		or original.Type ==
		"Beam"
		or original.Type ==
		"Trail" then

		object.Color =
			original.Color
	elseif original.Type == "Fire" then
		object.Color =
			original.Color

		object.SecondaryColor =
			original.SecondaryColor
	elseif original.Type == "Smoke" then
		object.Color =
			original.Color
	elseif original.Type == "Sparkles" then
		object.SparkleColor =
			original.Color
	end
end

local function enableVFXColorizer()
	for object in pairs(VFXInstances) do
		if object.Parent then
			setVFXColor(
				object,
				SETTINGS.VFXColor
			)
		else
			VFXInstances[object] = nil
		end
	end

	for _, object in ipairs(
		workspace:GetDescendants()
	) do
		if isVFXObject(object) then
			saveVFXOriginal(object)

			setVFXColor(
				object,
				SETTINGS.VFXColor
			)
		end
	end
end

local function disableVFXColorizer()
	for object in pairs(VFXInstances) do
		if object.Parent then
			restoreVFXColor(object)
		else
			VFXInstances[object] = nil
		end
	end
end

local function updateVFXColor()
	if not SETTINGS.CustomVFXColor then
		return
	end

	for object in pairs(VFXInstances) do
		if object.Parent then
			setVFXColor(
				object,
				SETTINGS.VFXColor
			)
		else
			VFXInstances[object] = nil
		end
	end
end

local function getCameraLockTarget()
	if CameraLockTarget
		and CameraLockTarget.Parent
		and not isTeammate(
			CameraLockTarget
		) then

		local character =
			CameraLockTarget.Character

		local humanoid =
			getHumanoid(character)

		local root =
			getRoot(character)

		if humanoid
			and root
			and humanoid.Health > 0 then

			return root
		end
	end

	local localRoot =
		getRoot(
			LocalPlayer.Character
		)

	if not localRoot then
		return nil
	end

	local nearestRoot
	local nearestDistance =
		math.huge

	for _, player in ipairs(
		Players:GetPlayers()
	) do
		if player ~= LocalPlayer
			and not isTeammate(player) then

			local character =
				player.Character

			local humanoid =
				getHumanoid(character)

			local root =
				getRoot(character)

			if humanoid
				and root
				and humanoid.Health > 0 then

				local offset =
					root.Position -
					localRoot.Position

				local horizontal =
					Vector3.new(
						offset.X,
						0,
						offset.Z
					)

				local distance =
					horizontal.Magnitude

				if distance <
					nearestDistance then

					nearestDistance =
						distance

					nearestRoot =
						root
				end
			end
		end
	end

	return nearestRoot
end

local function restoreCameraLock()
	local camera =
		getCamera()

	local humanoid =
		getHumanoid(
			LocalPlayer.Character
		)

	if humanoid
		and CameraLockAutoRotateSaved
			~= nil then

		humanoid.AutoRotate =
			CameraLockAutoRotateSaved
	end

	if camera then
		camera.CameraType =
			CameraLockCameraTypeSaved
			or Enum.CameraType.Custom

		if CameraLockCameraSubjectSaved then
			camera.CameraSubject =
				CameraLockCameraSubjectSaved
		elseif humanoid then
			camera.CameraSubject =
				humanoid
		end
	end

	CameraLockAutoRotateSaved = nil
	CameraLockCameraTypeSaved = nil
	CameraLockCameraSubjectSaved = nil
end

local function applyCameraLock()
	if not SETTINGS.CameraLock then
		restoreCameraLock()
		return
	end

	local camera =
		getCamera()

	local character =
		LocalPlayer.Character

	local humanoid =
		getHumanoid(character)

	local localRoot =
		getRoot(character)

	if not camera
		or not character
		or not humanoid
		or not localRoot
		or humanoid.Health <= 0 then

		return
	end

	local targetRoot =
		getCameraLockTarget()

	if not targetRoot then
		restoreCameraLock()
		return
	end

	if CameraLockAutoRotateSaved == nil then
		CameraLockAutoRotateSaved =
			humanoid.AutoRotate

		CameraLockCameraTypeSaved =
			camera.CameraType

		CameraLockCameraSubjectSaved =
			camera.CameraSubject
	end

	humanoid.AutoRotate = false

	local targetPosition =
		targetRoot.Position +
		Vector3.new(
			0,
			CameraLockTargetHeight,
			0
		)

	local flatTarget =
		Vector3.new(
			targetRoot.Position.X,
			localRoot.Position.Y,
			targetRoot.Position.Z
		)

	local direction =
		flatTarget -
		localRoot.Position

	if direction.Magnitude > 0.05 then
		localRoot.CFrame =
			CFrame.lookAt(
				localRoot.Position,
				flatTarget
			)
	end

	camera.CameraType =
		Enum.CameraType.Scriptable

	local cameraPosition =
		localRoot.Position +
		localRoot.CFrame.LookVector *
		-CameraLockDistance +
		Vector3.new(
			0,
			CameraLockHeight,
			0
		)

	camera.CFrame =
		CFrame.lookAt(
			cameraPosition,
			targetPosition
		)
end

local function applyLowGravity()
	if SETTINGS.LowGravity then
		workspace.Gravity =
			OriginalGravity *
			LOW_GRAVITY_MULTIPLIER
	else
		if math.abs(
			workspace.Gravity -
			(
				OriginalGravity *
				LOW_GRAVITY_MULTIPLIER
			)
		) < 0.01 then

			workspace.Gravity =
				OriginalGravity
		end
	end
end

local function applySpin(deltaTime)
	if not SETTINGS.Spin
		or SETTINGS.CameraLock then
		return
	end

	local character =
		LocalPlayer.Character

	local humanoid =
		getHumanoid(character)

	local root =
		getRoot(character)

	if not character
		or not humanoid
		or not root
		or humanoid.Health <= 0 then

		return
	end

	local speed =
		tonumber(
			SETTINGS.SpinSpeedValue
		)
		or 180

	if speed == 0 then
		return
	end

	SpinAngle =
		(
			SpinAngle +
			math.rad(speed) *
			deltaTime
		)
	%
	(math.pi * 2)

	local position =
		root.Position

	root.CFrame =
		CFrame.new(position) *
		CFrame.Angles(
			0,
			SpinAngle,
			0
		)
end

local function destroyESP(target)
	local data =
		EspObjects[target]

	if not data then
		return
	end

	for _, object in pairs(data) do
		if typeof(object) == "Instance" then
			if object.Parent then
				object:Destroy()
			end
		elseif typeof(object) == "table" then
			for _, item in ipairs(object) do
				if typeof(item) == "Instance"
					and item.Parent then

					item:Destroy()
				end
			end
		end
	end

	EspObjects[target] = nil
end

local function isValidNPC(model)
	if not model
		or not model:IsA("Model") then
		return false
	end

	if model == LocalPlayer.Character then
		return false
	end

	if Players:GetPlayerFromCharacter(
		model
	) then
		return false
	end

	return getHumanoid(model) ~= nil
		and getRoot(model) ~= nil
end

local function registerNPC(model)
	if not isValidNPC(model)
		or NpcModels[model] then

		return
	end

	NpcModels[model] = true
	createESP(
		model,
		false
	)
end

local function unregisterNPC(model)
	NpcModels[model] = nil
	destroyESP(model)
end

for _, object in ipairs(
	workspace:GetDescendants()
) do
	if object:IsA("Model") then
		registerNPC(object)
	end
end

workspace.DescendantAdded:Connect(
	function(object)
		if object:IsA("Model") then
			registerNPC(object)
		elseif object:IsA("Humanoid")
			or object.Name ==
				"HumanoidRootPart" then

			local model =
				object:FindFirstAncestorOfClass(
					"Model"
				)

			if model then
				task.defer(
					function()
						registerNPC(model)
					end
				)
			end
		end
	end
)

workspace.DescendantRemoving:Connect(
	function(object)
		if object:IsA("Model") then
			task.defer(
				function()
					if not isValidNPC(object) then
						unregisterNPC(
							object
						)
					end
				end
			)
		end
	end
)

local function getDistanceScale(distance)
	return math.clamp(
		1 /
			math.sqrt(
				math.max(
					distance,
					10
				) /
				18
			),
		0.5,
		1
	)
end

local function calculateScreenBox(
	character,
	camera
)
	local root =
		getRoot(character)

	if not root then
		return nil
	end

	local cf, size =
		character:GetBoundingBox()

	local half =
		size / 2

	local points = {
		cf * Vector3.new(
			-half.X,
			-half.Y,
			-half.Z
		),
		cf * Vector3.new(
			-half.X,
			-half.Y,
			half.Z
		),
		cf * Vector3.new(
			-half.X,
			half.Y,
			-half.Z
		),
		cf * Vector3.new(
			-half.X,
			half.Y,
			half.Z
		),
		cf * Vector3.new(
			half.X,
			-half.Y,
			-half.Z
		),
		cf * Vector3.new(
			half.X,
			-half.Y,
			half.Z
		),
		cf * Vector3.new(
			half.X,
			half.Y,
			-half.Z
		),
		cf * Vector3.new(
			half.X,
			half.Y,
			half.Z
		)
	}

	local minX = math.huge
	local minY = math.huge
	local maxX = -math.huge
	local maxY = -math.huge
	local visible = false

	for _, point in ipairs(points) do
		local screen =
			camera:WorldToViewportPoint(
				point
			)

		if screen.Z > 0 then
			visible = true

			minX =
				math.min(
					minX,
					screen.X
				)

			minY =
				math.min(
					minY,
					screen.Y
				)

			maxX =
				math.max(
					maxX,
					screen.X
				)

			maxY =
				math.max(
					maxY,
					screen.Y
				)
		end
	end

	if not visible then
		return nil
	end

	local rootScreen =
		camera:WorldToViewportPoint(
			root.Position
		)

	if rootScreen.Z <= 0 then
		return nil
	end

	local viewport =
		camera.ViewportSize

	if rootScreen.X < -250
		or rootScreen.X >
			viewport.X + 250
		or rootScreen.Y < -250
		or rootScreen.Y >
			viewport.Y + 250 then

		return nil
	end

	minX =
		math.clamp(
			minX,
			0,
			viewport.X
		)

	minY =
		math.clamp(
			minY,
			0,
			viewport.Y
		)

	maxX =
		math.clamp(
			maxX,
			0,
			viewport.X
		)

	maxY =
		math.clamp(
			maxY,
			0,
			viewport.Y
		)

	if maxX <= minX
		or maxY <= minY then

		return nil
	end

	return
		minX,
		minY,
		math.max(
			maxX - minX,
			4
		),
		math.max(
			maxY - minY,
			8
		)
end

local function getLinePoint(
	worldPosition,
	camera
)
	local screen =
		camera:WorldToViewportPoint(
			worldPosition
		)

	local viewport =
		camera.ViewportSize

	if screen.Z <= 0 then
		local cameraSpace =
			camera.CFrame:PointToObjectSpace(
				worldPosition
			)

		local direction =
			Vector2.new(
				cameraSpace.X,
				-cameraSpace.Y
			)

		if direction.Magnitude < 0.001 then
			direction =
				Vector2.new(
					0,
					1
				)
		end

		direction = direction.Unit

		local center =
			Vector2.new(
				viewport.X / 2,
				viewport.Y / 2
			)

		local half =
			Vector2.new(
				viewport.X / 2 - 4,
				viewport.Y / 2 - 4
			)

		local scaleX =
			half.X /
			math.max(
				math.abs(
					direction.X
				),
				0.0001
			)

		local scaleY =
			half.Y /
			math.max(
				math.abs(
					direction.Y
				),
				0.0001
			)

		local scale =
			math.min(
				scaleX,
				scaleY
			)

		local edge =
			center +
			direction *
			scale

		return Vector2.new(
			math.clamp(
				edge.X,
				0,
				viewport.X
			),
			math.clamp(
				edge.Y,
				0,
				viewport.Y
			)
		)
	end

	return Vector2.new(
		screen.X,
		screen.Y
	)
end

local function setLineBetween(
	line,
	startPoint,
	endPoint,
	thickness
)
	local difference =
		endPoint -
		startPoint

	local length =
		difference.Magnitude

	if length <= 0.5 then
		line.Visible = false
		return
	end

	local midpoint =
		startPoint +
		difference *
		0.5

	line.Position =
		UDim2.fromOffset(
			midpoint.X,
			midpoint.Y
		)

	line.Size =
		UDim2.fromOffset(
			length,
			thickness
		)

	line.AnchorPoint =
		Vector2.new(
			0.5,
			0.5
		)

	line.Rotation =
		math.deg(
			math.atan2(
				difference.Y,
				difference.X
			)
		)

	line.Visible = true
end

local function updateSkeleton(
	data,
	character,
	camera
)
	if not SETTINGS.Skeleton then
		data.Skeleton.Visible = false

		for _, line in ipairs(
			data.SkeletonLines
		) do
			line.Visible = false
		end

		return
	end

	local pairsToUse =
		character:FindFirstChild(
			"UpperTorso"
		)
		and SkeletonR15
		or SkeletonR6

	data.Skeleton.Visible = true

	for i, bone in ipairs(
		pairsToUse
	) do
		local partA =
			character:FindFirstChild(
				bone[1]
			)

		local partB =
			character:FindFirstChild(
				bone[2]
			)

		local line =
			data.SkeletonLines[i]

		if partA
			and partA:IsA("BasePart")
			and partB
			and partB:IsA("BasePart") then

			local pointA =
				camera:WorldToViewportPoint(
					partA.Position
				)

			local pointB =
				camera:WorldToViewportPoint(
					partB.Position
				)

			if pointA.Z > 0
				and pointB.Z > 0 then

				setLineBetween(
					line,
					Vector2.new(
						pointA.X,
						pointA.Y
					),
					Vector2.new(
						pointB.X,
						pointB.Y
					),
					2
				)
			else
				line.Visible = false
			end
		else
			line.Visible = false
		end
	end

	for i =
		#pairsToUse + 1,
		#data.SkeletonLines do

		data.SkeletonLines[i].Visible =
			false
	end
end

local function updateTracer(
	data,
	localRoot,
	targetRoot,
	camera
)
	if not SETTINGS.Lines
		or not localRoot
		or not targetRoot then

		data.Tracer.Visible = false
		return
	end

	local startPoint =
		getLinePoint(
			localRoot.Position,
			camera
		)

	local endPoint =
		getLinePoint(
			targetRoot.Position,
			camera
		)

	if not startPoint
		or not endPoint then

		data.Tracer.Visible = false
		return
	end

	setLineBetween(
		data.Tracer,
		startPoint,
		endPoint,
		1.5
	)
end

local function hideESP(data)
	data.Billboard.Enabled = false
	data.HealthBillboard.Enabled = false
	data.Box.Visible = false
	data.Highlight.Enabled = false
	data.Skeleton.Visible = false
	data.Tracer.Visible = false

	for _, line in ipairs(
		data.SkeletonLines
	) do
		line.Visible = false
	end
end

local function updateESP(target)
	local data =
		EspObjects[target]

	if not data then
		if typeof(target) ==
			"Instance"
			and target:IsA("Player") then

			createESP(
				target,
				true
			)
		elseif typeof(target) ==
			"Instance"
			and isValidNPC(target) then

			createESP(
				target,
				false
			)
		end

		data =
			EspObjects[target]
	end

	if not data then
		return
	end

	if data.IsPlayer
		and isTeammate(
			data.Target
		) then

		hideESP(data)
		return
	end

	local character =
		data.IsPlayer
		and data.Target.Character
		or data.Target

	local humanoid =
		getHumanoid(character)

	local root =
		getRoot(character)

	local camera =
		getCamera()

	if not character
		or not humanoid
		or not root
		or humanoid.Health <= 0
		or not camera then

		hideESP(data)
		return
	end

	if not data.IsPlayer
		and not SETTINGS.NPCESP then

		hideESP(data)
		return
	end

	local distance =
		(
			camera.CFrame.Position -
			root.Position
		).Magnitude

	if distance > ESP_MAX_DISTANCE then
		hideESP(data)
		return
	end

	local distanceScale =
		getDistanceScale(distance)

	data.Billboard.Adornee = root
	data.Billboard.Enabled =
		SETTINGS.Names
		or SETTINGS.Distance

	data.Billboard.Size =
		UDim2.fromOffset(
			math.max(
				90,
				180 *
					distanceScale
			),
			math.max(
				31,
				62 *
					distanceScale
			)
		)

	data.Billboard.StudsOffset =
		Vector3.new(
			0,
			2.9 +
				1.3 *
				distanceScale,
			0
		)

	data.NameLabel.Visible =
		SETTINGS.Names

	data.NameLabel.Text =
		data.IsPlayer
		and data.Target.DisplayName
		or data.Target.Name

	data.NameLabel.TextSize =
		math.max(
			10,
			14 *
				distanceScale
		)

	data.InfoLabel.Visible =
		SETTINGS.Distance

	data.InfoLabel.Text =
		string.format(
			"%d studs",
			math.floor(
				distance + 0.5
			)
		)

	data.InfoLabel.TextSize =
		math.max(
			9,
			11 *
				distanceScale
		)

	data.HealthBillboard.Adornee =
		root

	data.HealthBillboard.Enabled =
		SETTINGS.Health

	local cameraRight =
		camera.CFrame.RightVector

	data.HealthBillboard.StudsOffsetWorldSpace =
		-cameraRight *
		(
			2.25 *
			distanceScale
		)
		+
		Vector3.new(
			0,
			0.7 *
				distanceScale,
			0
		)

	data.HealthBillboard.Size =
		UDim2.fromOffset(
			math.max(
				12,
				16 *
					distanceScale
			),
			math.max(
				55,
				82 *
					distanceScale
			)
		)

	data.HealthBack.Size =
		UDim2.fromOffset(
			math.max(
				6,
				8 *
					distanceScale
			),
			math.max(
				45,
				72 *
					distanceScale
			)
		)

	data.HealthBack.Position =
		UDim2.new(
			0.5,
			-math.max(
				3,
				4 *
					distanceScale
			),
			0.5,
			-math.max(
				22,
				36 *
					distanceScale
			)
		)

	if SETTINGS.Health then
		local percent =
			math.clamp(
				humanoid.Health /
					math.max(
						humanoid.MaxHealth,
						1
					),
				0,
				1
			)

		data.HealthFill.Size =
			UDim2.new(
				1,
				0,
				percent,
				0
			)

		data.HealthFill.BackgroundColor3 =
			Color3.fromRGB(
				255 -
					math.floor(
						percent *
						180
					),
				75 +
					math.floor(
						percent *
						165
					),
				75
			)
	end

	if SETTINGS.Boxes then
		local x,
			y,
			width,
			height =
			calculateScreenBox(
				character,
				camera
			)

		if x then
			data.Box.Position =
				UDim2.fromOffset(
					math.floor(x),
					math.floor(y)
				)

			data.Box.Size =
				UDim2.fromOffset(
					math.floor(width),
					math.floor(height)
				)

			local thickness =
				math.clamp(
					math.floor(
						math.min(
							width,
							height
						) *
						0.025
					),
					1,
					3
				)

			data.Top.Size =
				UDim2.new(
					1,
					0,
					0,
					thickness
				)

			data.Top.Position =
				UDim2.fromOffset(
					0,
					0
				)

			data.Bottom.Size =
				UDim2.new(
					1,
					0,
					0,
					thickness
				)

			data.Bottom.Position =
				UDim2.new(
					0,
					0,
					1,
					-thickness
				)

			data.Left.Size =
				UDim2.new(
					0,
					thickness,
					1,
					0
				)

			data.Left.Position =
				UDim2.fromOffset(
					0,
					0
				)

			data.Right.Size =
				UDim2.new(
					0,
					thickness,
					1,
					0
				)

			data.Right.Position =
				UDim2.new(
					1,
					-thickness,
					0,
					0
				)

			data.Box.Visible = true
		else
			data.Box.Visible = false
		end
	else
		data.Box.Visible = false
	end

	data.Highlight.Adornee =
		character

	data.Highlight.Enabled =
		SETTINGS.Outline

	updateSkeleton(
		data,
		character,
		camera
	)

	updateTracer(
		data,
		getRoot(
			LocalPlayer.Character
		),
		root,
		camera
	)
end

local function updateAllESP()
	for _, player in ipairs(
		Players:GetPlayers()
	) do
		if player ~= LocalPlayer then
			updateESP(player)
		end
	end

	for model in pairs(NpcModels) do
		if model.Parent then
			updateESP(model)
		else
			unregisterNPC(model)
		end
	end
end

local function makeVignette(
	size,
	position,
	rotation
)
	local frame =
		new(
			"Frame",
			{
				Size = size,
				Position = position,
				BackgroundColor3 =
					Color3.fromRGB(
						95,
						125,
						255
					),
				BackgroundTransparency = 1,
				BorderSizePixel = 0,
				Active = false,
				ZIndex = 901
			},
			FXGui
		)

	new(
		"UIGradient",
		{
			Rotation = rotation,
			Transparency =
				NumberSequence.new(
					{
						NumberSequenceKeypoint.new(
							0,
							0.05
						),
						NumberSequenceKeypoint.new(
							1,
							1
						)
					}
				)
		},
		frame
	)

	return frame
end

local function connectHealth(
	character
)
	local humanoid =
		getHumanoid(character)

	if not humanoid then
		return
	end

	LastHealth =
		humanoid.Health

	humanoid.HealthChanged:Connect(
		function(health)
			if LastHealth
				and health < LastHealth
				and SETTINGS.DamageFX then

				DamageFlash.BackgroundColor3 =
					Color3.fromRGB(
						255,
						45,
						45
					)

				DamageFlash.BackgroundTransparency =
					SETTINGS.ReducedMotion
					and 0.85
					or 0.72

				TweenService:Create(
					DamageFlash,
					TweenInfo.new(
						SETTINGS.ReducedMotion
						and 0.08
						or 0.3,
						Enum.EasingStyle.Quad,
						Enum.EasingDirection.Out
					),
					{
						BackgroundTransparency = 1
					}
				):Play()
			end

			LastHealth =
				health
		end
	)
end

local function updateScreenFX()
	local camera =
		getCamera()

	if camera then
		local position =
			camera.CFrame.Position

		local look =
			camera.CFrame.LookVector

		if CameraPositionLast
			and CameraLookLast then

			local positionDelta =
				(
					position -
					CameraPositionLast
				).Magnitude

			local lookDelta =
				(
					look -
					CameraLookLast
				).Magnitude

			CameraMotion =
				math.clamp(
					positionDelta *
						0.45 +
						lookDelta *
						14,
					0,
					1
				)
		end

		CameraPositionLast =
			position

		CameraLookLast =
			look
	end

	if SETTINGS.Vignette then
		local transparency =
			SETTINGS.ExtremeExtremeColors
			and 0.04
			or SETTINGS.ExtremeColors
			and 0.08
			or 0.18

		TopVignette.BackgroundTransparency =
			transparency

		BottomVignette.BackgroundTransparency =
			transparency

		LeftVignette.BackgroundTransparency =
			transparency

		RightVignette.BackgroundTransparency =
			transparency
	else
		TopVignette.BackgroundTransparency = 1
		BottomVignette.BackgroundTransparency = 1
		LeftVignette.BackgroundTransparency = 1
		RightVignette.BackgroundTransparency = 1
	end

	local humanoid =
		getHumanoid(
			LocalPlayer.Character
		)

	if SETTINGS.LowHealth
		and humanoid
		and humanoid.MaxHealth > 0
		and humanoid.Health > 0 then

		local percent =
			humanoid.Health /
			humanoid.MaxHealth

		if percent <= 0.35 then
			local intensity =
				1 -
				math.clamp(
					percent /
						0.35,
					0,
					1
				)

			if SETTINGS.ReducedMotion then
				LowHealthOverlay.BackgroundTransparency =
					0.92
			else
				local pulse =
					(
						math.sin(
							os.clock() *
							5
						) + 1
					)
					*
					0.5

				local strength =
					0.1

				if SETTINGS.ColorfulVFX then
					strength += 0.08
				end

				if SETTINGS.ExtremeColors then
					strength += 0.12
				end

				if SETTINGS.ExtremeExtremeColors then
					strength += 0.2
				end

				LowHealthOverlay.BackgroundTransparency =
					0.97
					-
					intensity *
					strength
					-
					pulse *
					intensity *
					0.03
			end
		else
			LowHealthOverlay.BackgroundTransparency =
				1
		end
	else
		LowHealthOverlay.BackgroundTransparency =
			1
	end

	if SETTINGS.MotionBlur
		and camera
		and not SETTINGS.ReducedMotion then

		local blurStrength = 7

		if SETTINGS.ColorfulVFX then
			blurStrength += 3
		end

		if SETTINGS.ExtremeColors then
			blurStrength += 6
		end

		if SETTINGS.ExtremeExtremeColors then
			blurStrength += 10
		end

		Blur.Size =
			CameraMotion *
			blurStrength
	else
		Blur.Size = 0
	end

	local saturation = 0
	local contrast = 0
	local brightness = 0

	if SETTINGS.ColorBoost then
		saturation += 0.08
		contrast += 0.05
	end

	if SETTINGS.ColorfulVFX then
		saturation += 0.4
		contrast += 0.12
	end

	if SETTINGS.ExtremeColors then
		saturation += 0.8
		contrast += 0.35
		brightness += 0.04
	end

	if SETTINGS.ExtremeExtremeColors then
		saturation = 1
		contrast = 1
		brightness = 0.08
	end

	ColorCorrection.Saturation =
		math.clamp(
			saturation,
			-1,
			1
		)

	ColorCorrection.Contrast =
		math.clamp(
			contrast,
			-1,
			1
		)

	ColorCorrection.Brightness =
		math.clamp(
			brightness,
			-1,
			1
		)

	ColorCorrection.TintColor =
		Color3.new(
			1,
			1,
			1
		)

	ColorCorrection.Enabled =
		SETTINGS.ColorBoost
		or SETTINGS.ColorfulVFX
		or SETTINGS.ExtremeColors
		or SETTINGS.ExtremeExtremeColors

	if SETTINGS.ExtremeExtremeColors then
		ExtremeBloom.Intensity = 2.5
		ExtremeBloom.Size = 32
		ExtremeBloom.Threshold = 0.7
	elseif SETTINGS.ExtremeColors then
		ExtremeBloom.Intensity = 1.2
		ExtremeBloom.Size = 28
		ExtremeBloom.Threshold = 0.8
	elseif SETTINGS.ColorfulVFX then
		ExtremeBloom.Intensity = 0.45
		ExtremeBloom.Size = 24
		ExtremeBloom.Threshold = 1
	else
		ExtremeBloom.Intensity = 0
		ExtremeBloom.Size = 24
		ExtremeBloom.Threshold = 1
	end

	ExtremeBloom.Enabled =
		SETTINGS.ColorfulVFX
		or SETTINGS.ExtremeColors
		or SETTINGS.ExtremeExtremeColors

	if SETTINGS.NightMode then
		Lighting.ClockTime = 0.5
		Lighting.Brightness = 1.2
		Lighting.Ambient =
			Color3.fromRGB(
				18,
				22,
				45
			)
		Lighting.OutdoorAmbient =
			Color3.fromRGB(
				10,
				15,
				35
			)
		Lighting.ExposureCompensation =
			-0.15
	else
		Lighting.ClockTime =
			OriginalLighting.ClockTime

		Lighting.Brightness =
			OriginalLighting.Brightness

		Lighting.Ambient =
			OriginalLighting.Ambient

		Lighting.OutdoorAmbient =
			OriginalLighting.OutdoorAmbient

		Lighting.ExposureCompensation =
			OriginalLighting.ExposureCompensation
	end

	HorrorCorrection.Enabled =
		SETTINGS.HorrorMode

	if SETTINGS.HorrorMode then
		HorrorCorrection.Brightness =
			-0.18

		HorrorCorrection.Contrast =
			0.4

		HorrorCorrection.Saturation =
			0.05

		HorrorCorrection.Saturation = 0

local function updateResponsiveScale()
	local viewport =
		getViewportSize()

	if viewport.X <= 0
		or viewport.Y <= 0 then

		return
	end

	local panelScale =
		math.clamp(
			math.min(
				viewport.X / 700,
				viewport.Y / 700
			),
			0.68,
			1.12
		)

	if UserInputService.TouchEnabled
		and not UserInputService.KeyboardEnabled then

		panelScale =
			math.clamp(
				panelScale * 1.03,
				0.68,
				1.12
			)
	end

	MainScale.Scale =
		panelScale

	OpenButtonScale.Scale =
		math.clamp(
			math.min(
				viewport.X / 800,
				viewport.Y / 800
			),
			0.82,
			1.15
		)
end

local function connectCameraScaling()
	if currentCameraConnection then
		currentCameraConnection:Disconnect()
		currentCameraConnection = nil
	end

	local camera =
		getCamera()

	if camera then
		currentCameraConnection =
			camera:GetPropertyChangedSignal(
				"ViewportSize"
			):Connect(
				updateResponsiveScale
			)
	end

	updateResponsiveScale()
end

workspace:GetPropertyChangedSignal(
	"CurrentCamera"
):Connect(
	function()
		connectCameraScaling()

		if SETTINGS.CameraLock then
			CameraLockAutoRotateSaved = nil
			CameraLockCameraTypeSaved = nil
			CameraLockCameraSubjectSaved = nil
		end
	end
)

connectCameraScaling()

Header.InputBegan:Connect(
	function(input)
		if input.UserInputType ~=
			Enum.UserInputType.MouseButton1
			and input.UserInputType ~=
			Enum.UserInputType.Touch then

			return
		end

		draggingMain = true
		dragStart =
			input.Position

		startPos =
			Main.Position
	end
)

UserInputService.InputChanged:Connect(
	function(input)
		if not draggingMain then
			return
		end

		if input.UserInputType ~=
			Enum.UserInputType.MouseMovement
			and input.UserInputType ~=
			Enum.UserInputType.Touch then

			return
		end

		local delta =
			input.Position -
			dragStart

		Main.Position =
			UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset +
					delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset +
					delta.Y
			)
	end
)

UserInputService.InputEnded:Connect(
	function(input)
		if input.UserInputType ==
			Enum.UserInputType.MouseButton1
			or input.UserInputType ==
			Enum.UserInputType.Touch then

			draggingMain = false
		end
	end
)

UserInputService.JumpRequest:Connect(
	function()
		if not SETTINGS.InfiniteJump then
			return
		end

		local character =
			LocalPlayer.Character

		local humanoid =
			getHumanoid(character)

		if not humanoid
			or humanoid.Health <= 0 then

			return
		end

		humanoid.Jump = true

		humanoid:ChangeState(
			Enum.HumanoidStateType.Jumping
		)
	end
)

local function getCreatorAvatar()
	local success, userId =
		pcall(
			function()
				return Players:GetUserIdFromNameAsync(
					"Twitterboy_112"
				)
			end
		)

	if not success
		or not userId then

		return nil
	end

	local success2, image =
		pcall(
			function()
				return Players:GetUserThumbnailAsync(
					userId,
					Enum.ThumbnailType.AvatarBust,
					Enum.ThumbnailSize.Size420x420
				)
			end
		)

	if success2
		and image then

		return image
	end

	return
		"rbxthumb://type=AvatarBust&id="
		.. tostring(userId)
		.. "&w=420&h=420"
end
